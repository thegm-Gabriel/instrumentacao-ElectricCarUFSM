cmake_minimum_required(VERSION 3.16.0)
set(IDF_TARGET "esp32c3" CACHE STRING "Alvo do satélite da Equipe")
if(NOT IDF_TARGET STREQUAL "esp32c3")
    message(FATAL_ERROR "O satélite da Equipe requer ESP32-C3 SuperMini Plus.")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
if(DEFINED ENV{VERSAO_FIRMWARE})
    set(PROJECT_VER "$ENV{VERSAO_FIRMWARE}")
else()
    set(PROJECT_VER "0.1.0")
endif()
project(esp32c3_equipe)
